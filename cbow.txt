import re
import numpy as np
import tensorflow as tf
from tensorflow.keras import layers, models

# -----------------------------
# 1. Create Corpus
# -----------------------------

corpus = [
    "the quick brown fox jumps over the lazy dog",
    "the quick brown dog jumps over the lazy fox",
    "the cat sits on the mat",
    "the dog sits on the rug",
    "machine learning learns patterns from data",
    "deep learning models learn patterns",
    "word embeddings represent words as vectors",
    "similar words occur in similar contexts",
    "the model learns word context",
    "machine learning uses data and patterns",
    "deep learning uses neural networks",
    "the fox jumps over the dog",
    "the cat and dog are animals",
    "word context helps predict words",
    "neural networks learn useful features"
]

# Tokenization
tokens = [re.findall(r"[a-z]+", s.lower()) for s in corpus]

# Create vocabulary
vocab = sorted(set(w for sentence in tokens for w in sentence))

# Word-to-ID and ID-to-word mappings
word2id = {w: i for i, w in enumerate(vocab)}
id2word = {i: w for w, i in word2id.items()}

print("Vocabulary size:", len(vocab))
print("Sample tokens:", tokens[0])


# -----------------------------
# 2. Create Context-Target Pairs
# -----------------------------

window = 2
contexts, targets = [], []

for sentence in tokens:
    for i in range(window, len(sentence) - window):
        context = sentence[i-2:i] + sentence[i+1:i+3]

        contexts.append([word2id[w] for w in context])
        targets.append(word2id[sentence[i]])

# Convert to NumPy arrays
X = np.array(contexts)
y = np.array(targets)

print("Training examples:", len(X))
print("Context words:", [id2word[i] for i in X[0]])
print("Target word:", id2word[y[0]])


# -----------------------------
# 3. Build Neural Network Model
# -----------------------------

model = models.Sequential([
    layers.Input(shape=(4,)),
    layers.Embedding(len(vocab), 16),
    layers.GlobalAveragePooling1D(),
    layers.Dense(len(vocab), activation="softmax")
])


# -----------------------------
# 4. Compile Model
# -----------------------------

model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)


# -----------------------------
# 5. Train Model
# -----------------------------

history = model.fit(
    X,
    y,
    epochs=200,
    batch_size=4,
    verbose=0
)

print("Model training completed!")
print("Final accuracy:", round(history.history["accuracy"][-1], 3))


# -----------------------------
# 6. Predict Target Word
# -----------------------------

i = 0

probabilities = model.predict(
    X[i:i+1],
    verbose=0
)[0]

predicted = np.argmax(probabilities)

print("Context:", [id2word[int(w)] for w in X[i]])
print("Actual target:", id2word[y[i]])
print("Predicted target:", id2word[predicted])


# -----------------------------
# 7. Display Top 3 Predictions
# -----------------------------

print("\nTop predictions:")

for j in np.argsort(probabilities)[-3:][::-1]:
    print(
        id2word[j],
        round(float(probabilities[j]), 3)
    )

import matplotlib.pyplot as plt

%matplotlib inline


# Check NLTK resources
for resource in ["tokenizers/punkt", "tokenizers/punkt_tab", "corpora/stopwords"]:
    try:
        nltk.data.find(resource)
        print(resource, "available")
    except LookupError:
        print(resource, "missing. Download it before going offline.")


# Input text
sentences = """Machine learning helps computers learn from data.
Students use Python to build intelligent applications.
Neural networks can recognize patterns in images and text.
Deep learning models require training data and computing power.
Natural language processing allows computers to understand human language.
Word embeddings represent words as numerical vectors.
Artificial intelligence is transforming healthcare, education, and finance.
Researchers continuously improve machine learning algorithms."""

print(sentences)


# Remove special characters
sentences = re.sub(r"[^A-Za-z]+", " ", sentences)
sentences = re.sub(r"\b\w\b", " ", sentences).strip()

print(sentences)


# Convert to lowercase and tokenize
sentences = sentences.lower()

all_sent = nltk.sent_tokenize(sentences)
all_words = [nltk.word_tokenize(s) for s in all_sent]

print(all_words)


# Remove stop words
stop_words = set(stopwords.words("english"))

data = [
    [w for w in sentence if w not in stop_words]
    for sentence in all_words
]

data1 = data[0]

print(data)

    sentences=data,
    min_count=1,
    vector_size=52,
    window=5,
    workers=1,
    sg=0,
    seed=42,
    epochs=100
)

print("Word2Vec model trained.")
